using System;

/// <summary>Encloses generated water coverage in one bank-hidden convex polygon per chunk.</summary>
internal static class SurfaceWaterGeometry
{
	[System.Runtime.InteropServices.StructLayout( System.Runtime.InteropServices.LayoutKind.Sequential )]
	internal struct WaterVertex
	{
		[VertexLayout.Position] public Vector3 Position;
		public WaterVertex( Vector3 position ) { Position = position; }
	}
	internal sealed record Chunk( GpuSdfDescriptor Descriptor, GeneratedWaterCells Cells, WaterVertex[] Vertices, double Milliseconds )
	{
		public long CompletedTimestamp { get; } = System.Diagnostics.Stopwatch.GetTimestamp();
	}

	public static WaterVertex[] Build( GeneratedWaterCells cells )
	{
		const int size = GeneratedWaterCells.CellsPerAxis;
		// At each column boundary only the lowest/highest covered corners can
		// belong to the hull. These points are already sorted by X, then Y.
		Span<Vector2> points = stackalloc Vector2[(size + 1) * 2];
		var pointCount = 0;
		var previousLow = size;
		var previousHigh = -1;
		for ( var x = 0; x <= size; x++ )
		{
			var low = size;
			var high = -1;
			if ( x < size )
			{
				for ( var y = 0; y < size; y++ )
				{
					if ( !cells.SurfaceCoverage[x + size * y] ) continue;
					low = Math.Min( low, y );
					high = y;
				}
			}
			var boundaryLow = Math.Min( previousLow, low );
			var boundaryHigh = Math.Max( previousHigh, high );
			if ( boundaryHigh >= 0 )
			{
				points[pointCount++] = new Vector2( x, boundaryLow );
				points[pointCount++] = new Vector2( x, boundaryHigh + 1 );
			}
			previousLow = low;
			previousHigh = high;
		}
		if ( pointCount == 0 ) return Array.Empty<WaterVertex>();
		// Monotone chains remove collinear/inside corners using exact small-grid
		// arithmetic. The convex enclosure stays inside this chunk's footprint.
		Span<Vector2> hull = stackalloc Vector2[points.Length * 2];
		var count = 0;
		for ( var pass = 0; pass < 2; pass++ )
		{
			var minimumCount = pass == 0 ? 2 : count + 1;
			for ( var step = pass; step < pointCount; step++ )
			{
				var point = points[pass == 0 ? step : pointCount - 1 - step];
				while ( count >= minimumCount )
				{
					var a = hull[count - 1] - hull[count - 2];
					var b = point - hull[count - 2];
					if ( a.x * b.y - a.y * b.x > 0f ) break;
					count--;
				}
				hull[count++] = point;
			}
		}
		count--; // The upper chain closes on the first point.
		var vertices = new WaterVertex[(count - 2) * 3];
		var origin = cells.Bounds.Minimum;
		origin.z = cells.SeaLevel;
		var first = origin + new Vector3( hull[0].x * cells.CellSize, hull[0].y * cells.CellSize, 0f );
		var previous = origin + new Vector3( hull[1].x * cells.CellSize, hull[1].y * cells.CellSize, 0f );
		for ( var index = 2; index < count; index++ )
		{
			var current = origin + new Vector3( hull[index].x * cells.CellSize, hull[index].y * cells.CellSize, 0f );
			var output = (index - 2) * 3;
			vertices[output] = new WaterVertex( first );
			vertices[output + 1] = new WaterVertex( previous );
			vertices[output + 2] = new WaterVertex( current );
			previous = current;
		}
		return vertices;
	}
}
