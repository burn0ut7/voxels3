using System;

/// <summary>Encloses generated water coverage in one bank-hidden quad per chunk.</summary>
internal static class SurfaceWaterGeometry
{
	[System.Runtime.InteropServices.StructLayout( System.Runtime.InteropServices.LayoutKind.Sequential )]
	internal struct WaterVertex
	{
		[VertexLayout.Position] public Vector3 Position;
		public WaterVertex( Vector3 position ) { Position = position; }
	}
	internal sealed record Chunk( GpuSdfDescriptor Descriptor, GeneratedWaterCells Cells, WaterVertex[] Vertices, double Milliseconds )
	{
		public long CompletedTimestamp { get; } = System.Diagnostics.Stopwatch.GetTimestamp();
	}

	public static WaterVertex[] Build( GeneratedWaterCells cells )
	{
		const int size = GeneratedWaterCells.CellsPerAxis;
		var minimumX = 0;
		var minimumY = 0;
		var maximumX = size;
		var maximumY = size;
		if ( !cells.UniformlyWet )
		{
			minimumX = minimumY = size;
			maximumX = maximumY = 0;
			for ( var y = 0; y < size; y++ )
			for ( var x = 0; x < size; x++ )
			{
				if ( !cells.SurfaceCoverage[x + size * y] ) continue;
				minimumX = Math.Min( minimumX, x );
				minimumY = Math.Min( minimumY, y );
				maximumX = Math.Max( maximumX, x + 1 );
				maximumY = Math.Max( maximumY, y + 1 );
			}
			if ( maximumX == 0 ) return Array.Empty<WaterVertex>();
		}
		var origin = cells.Bounds.Minimum;
		origin.z = cells.SeaLevel;
		var first = new WaterVertex( origin + new Vector3( minimumX * cells.CellSize, minimumY * cells.CellSize, 0f ) );
		var right = new WaterVertex( origin + new Vector3( maximumX * cells.CellSize, minimumY * cells.CellSize, 0f ) );
		var diagonal = new WaterVertex( origin + new Vector3( maximumX * cells.CellSize, maximumY * cells.CellSize, 0f ) );
		var up = new WaterVertex( origin + new Vector3( minimumX * cells.CellSize, maximumY * cells.CellSize, 0f ) );
		return new WaterVertex[] { first, right, diagonal, first, diagonal, up };
	}
}
